1. Swap Two Numbers With and Without Temporary Variable

Aim

To swap two numbers using a temporary variable and without using a temporary variable.

Source Code

#include <iostream>
using namespace std;

int main() {
    int a, b, temp;
    cout << "Enter two numbers: ";
    cin >> a >> b;

    temp = a;
    a = b;
    b = temp;
    cout << "Using temporary variable: " << a << " " << b << endl;

    a = a + b;
    b = a - b;
    a = a - b;
    cout << "Without temporary variable: " << a << " " << b;

    return 0;
}

Sample Output

Enter two numbers: 10 20
Using temporary variable: 20 10
Without temporary variable: 10 20

Result

Thus, two numbers are swapped using both methods.

2. Convert Fahrenheit to Celsius and Celsius to Fahrenheit

Aim

To convert temperature from Fahrenheit to Celsius and Celsius to Fahrenheit.

Source Code

#include <iostream>
using namespace std;

int main() {
    float c, f;
    cout << "Enter temperature in Celsius: ";
    cin >> c;
    f = (c * 9 / 5) + 32;
    cout << "Fahrenheit = " << f << endl;

    cout << "Enter temperature in Fahrenheit: ";
    cin >> f;
    c = (f - 32) * 5 / 9;
    cout << "Celsius = " << c;

    return 0;
}

Sample Output

Enter temperature in Celsius: 100
Fahrenheit = 212
Enter temperature in Fahrenheit: 98.6
Celsius = 37

Result

Thus, temperature conversion is performed successfully.

4. Simple Calculator Using Functions

Aim

To perform addition, subtraction, multiplication and division using functions and switch statement.

Source Code

#include <iostream>
using namespace std;

float add(float a, float b) { return a+b; }
float sub(float a, float b) { return a-b; }
float mul(float a, float b) { return a*b; }
float divide(float a, float b) { return a/b; }

int main() {
    float a, b;
    int choice;
    cout << "Enter two numbers: ";
    cin >> a >> b;

    cout << "1.Add  2.Subtract  3.Multiply  4.Divide\n";
    cout << "Enter choice: ";
    cin >> choice;

    switch(choice) {
        case 1: cout << add(a,b); break;
        case 2: cout << sub(a,b); break;
        case 3: cout << mul(a,b); break;
        case 4:
            if(b != 0) cout << divide(a,b);
            else cout << "Division by zero not allowed";
            break;
        default: cout << "Invalid choice";
    }
    return 0;
}

Sample Output

Enter two numbers: 20 5
1.Add  2.Subtract  3.Multiply  4.Divide
Enter choice: 3
100

Result

Thus, a simple calculator is implemented using functions.

6. Check Whether a Number is Palindrome

Aim

To check whether a given number is a palindrome or not.

Source Code

#include <iostream>
using namespace std;

int main() {
    int n, original, reverse=0, digit;
    cout << "Enter a number: ";
    cin >> n;
    original = n;

    while(n > 0) {
        digit = n % 10;
        reverse = reverse * 10 + digit;
        n = n / 10;
    }

    if(original == reverse)
        cout << "Palindrome number";
    else
        cout << "Not a palindrome number";

    return 0;
}

Sample Output

Enter a number: 121
Palindrome number

Result

Thus, the given number is checked for palindrome.

8. Sum of Principal, Lower and Upper Diagonal Elements of a Matrix

Aim

To calculate the sums of principal diagonal, lower triangle and upper triangle elements of a square matrix.

Source Code

#include <iostream>
using namespace std;

int main() {
    int a[10][10], n;
    int principal=0, lower=0, upper=0;

    cout << "Enter order of matrix: ";
    cin >> n;

    cout << "Enter matrix elements:\n";
    for(int i=0; i<n; i++)
        for(int j=0; j<n; j++)
            cin >> a[i][j];

    for(int i=0; i<n; i++) {
        for(int j=0; j<n; j++) {
            if(i == j) principal += a[i][j];
            else if(i > j) lower += a[i][j];
            else upper += a[i][j];
        }
    }

    cout << "Principal diagonal sum = " << principal << endl;
    cout << "Lower triangle sum = " << lower << endl;
    cout << "Upper triangle sum = " << upper;

    return 0;
}

Sample Output

Enter order of matrix: 3
Enter matrix elements:
1 2 3
4 5 6
7 8 9
Principal diagonal sum = 15
Lower triangle sum = 19
Upper triangle sum = 11

Result

Thus, the three matrix sums are calculated successfully. Lower and upper sums exclude the principal diagonal.

9. Reverse a String Without Using Built-in Function

Aim

To reverse a given string without using a built-in reverse function.

Source Code

#include <iostream>
#include <cstring>
using namespace std;

int main() {
    char str[100];
    int len=0;

    cout << "Enter a string: ";
    cin.getline(str,100);

    while(str[len] != '\0')
        len++;

    cout << "Reversed string: ";
    for(int i=len-1; i>=0; i--)
        cout << str[i];

    return 0;
}

Sample Output

Enter a string: STUDENT
Reversed string: TNEDUTS

Result

Thus, the string is reversed without using a built-in reverse function.

11. Call by Value and Call by Reference

Aim

To demonstrate call by value and call by reference in C++.

Source Code

#include <iostream>
using namespace std;

void callByValue(int x) {
    x = x + 10;
    cout << "Inside call by value: " << x << endl;
}

void callByReference(int &x) {
    x = x + 10;
    cout << "Inside call by reference: " << x << endl;
}

int main() {
    int a=10, b=10;

    callByValue(a);
    cout << "After call by value: " << a << endl;

    callByReference(b);
    cout << "After call by reference: " << b;

    return 0;
}

Sample Output

Inside call by value: 20
After call by value: 10
Inside call by reference: 20
After call by reference: 20

Result

Thus, call by value and call by reference are demonstrated.

12. Factorial Using Recursion

Aim

To find the factorial of a number using a recursive function.

Source Code

#include <iostream>
using namespace std;

int factorial(int n) {
    if(n <= 1)
        return 1;
    return n * factorial(n-1);
}

int main() {
    int n;
    cout << "Enter a number: ";
    cin >> n;

    if(n < 0)
        cout << "Factorial is not defined for negative numbers";
    else
        cout << "Factorial = " << factorial(n);

    return 0;
}

Sample Output

Enter a number: 5
Factorial = 120

Result

Thus, factorial is calculated using recursion.

15. Friend Function

Aim

To demonstrate the use of a friend function to access private class members.

Source Code

#include <iostream>
using namespace std;

class Number {
    int a, b;

public:
    void getData() {
        cout << "Enter two numbers: ";
        cin >> a >> b;
    }

    friend int sum(Number n);
};

int sum(Number n) {
    return n.a + n.b;
}

int main() {
    Number obj;
    obj.getData();
    cout << "Sum = " << sum(obj);
    return 0;
}

Sample Output

Enter two numbers: 15 25
Sum = 40

Result

Thus, a friend function accesses private data members of a class.

To demonstrate multilevel inheritance using three classes.

18. Operator Overloading

Aim

To demonstrate binary operator overloading using the plus (+) operator.

Source Code

#include <iostream>
using namespace std;

class Number {
    int value;

public:
    Number(int v=0) {
        value = v;
    }

    Number operator +(Number obj) {
        Number temp;
        temp.value = value + obj.value;
        return temp;
    }

    void display() {
        cout << value;
    }
};

int main() {
    Number n1(10), n2(20), n3;
    n3 = n1 + n2;

    cout << "First number = ";
    n1.display();

    cout << "\nSecond number = ";
    n2.display();

    cout << "\nSum = ";
    n3.display();

    return 0;
}

Sample Output

First number = 10
Second number = 20
Sum = 30

Result

Thus, the plus operator is overloaded to add two objects.

Practical Examination Preparation

Important Practical Viva Questions

1. What is C++?
C++ is a general-purpose programming language that supports procedural and object-oriented programming.

2. What is a class?
A class is a user-defined data type that contains data members and member functions.

3. What is an object?
An object is an instance of a class.

4. What is recursion?
Recursion is a process in which a function calls itself.

5. What is function overloading?
Defining multiple functions with the same name but different parameter lists.

6. What is a friend function?
A non-member function that is allowed to access private and protected members of a class.

7. What is inheritance?
Inheritance allows one class to acquire the properties and functions of another class.

8. What is operator overloading?
Giving a special meaning to an existing operator for user-defined data types.

9. What is call by value?
A copy of the argument is passed to the function.

10. What is call by reference?
A reference to the original variable is passed to the function.
