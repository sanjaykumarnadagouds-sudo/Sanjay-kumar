1. Program to Find GCD of Two Numbers

Aim: To write a C++ program to find the Greatest Common Divisor (GCD) of two numbers.

Source Code

#include <iostream>
using namespace std;

int main() {
    int a, b, temp;
    cout << "Enter two numbers: ";
    cin >> a >> b;

    int x = a, y = b;

    while (y != 0) {
        temp = y;
        y = x % y;
        x = temp;
    }

    cout << "GCD = " << x;
    return 0;
}

Sample Output

Enter two numbers: 24 36 GCD = 12

Result: The GCD of two numbers is calculated successfully.

2. Program to Implement Tower of Hanoi

Aim: To solve the Tower of Hanoi problem using recursion.

Source Code

#include <iostream>
using namespace std;

void tower(int n, char source, char auxiliary, char destination) {
    if (n == 1) {
        cout << "Move disk 1 from "
             << source << " to " << destination << endl;
        return;
    }

    tower(n - 1, source, destination, auxiliary);

    cout << "Move disk " << n << " from "
         << source << " to " << destination << endl;

    tower(n - 1, auxiliary, source, destination);
}

int main() {
    int n;
    cout << "Enter number of disks: ";
    cin >> n;

    tower(n, 'A', 'B', 'C');
    return 0;
}

Sample Output

Result: Tower of Hanoi is implemented successfully using recursion.

5. Program to Perform Stack Operations

Aim: To implement stack operations using an array.

Source Code

#include <iostream>
using namespace std;

#define MAX 5

int stack[MAX];
int top = -1;

void push(int value) {
    if (top == MAX - 1)
        cout << "Stack Overflow\n";
    else {
        stack[++top] = value;
        cout << "Inserted: " << value << endl;
    }
}

void pop() {
    if (top == -1)
        cout << "Stack Underflow\n";
    else
        cout << "Deleted: " << stack[top--] << endl;
}

void display() {
    if (top == -1) {
        cout << "Stack is empty\n";
        return;
    }

    cout << "Stack elements: ";
    for (int i = top; i >= 0; i--)
        cout << stack[i] << " ";
    cout << endl;
}

int main() {
    int choice, value;

    do {
        cout << "\n1.Push 2.Pop 3.Display 4.Exit\n";
        cout << "Enter choice: ";
        cin >> choice;

        switch (choice) {
            case 1:
                cout << "Enter value: ";
                cin >> value;
                push(value);
                break;
            case 2:
                pop();
                break;
            case 3:
                display();
                break;
            case 4:
                cout << "Exiting...";
                break;
            default:
                cout << "Invalid choice\n";
        }
    } while (choice != 4);

    return 0;
}

6. Program to Perform Linear Queue Operations

Aim: To implement insertion, deletion and display operations in a linear queue using an array.

Source Code

#include <iostream>
using namespace std;

#define MAX 5

int queueArray[MAX];
int front = -1, rear = -1;

void enqueue(int value) {
    if (rear == MAX - 1) {
        cout << "Queue Overflow\n";
    } else {
        if (front == -1)
            front = 0;

        queueArray[++rear] = value;
        cout << "Inserted: " << value << endl;
    }
}

void dequeue() {
    if (front == -1 || front > rear) {
        cout << "Queue Underflow\n";
    } else {
        cout << "Deleted: " << queueArray[front++] << endl;

        if (front > rear)
            front = rear = -1;
    }
}

void display() {
    if (front == -1) {
        cout << "Queue is empty\n";
        return;
    }

    cout << "Queue elements: ";
    for (int i = front; i <= rear; i++)
        cout << queueArray[i] << " ";
    cout << endl;
}

int main() {
    int choice, value;

    do {
        cout << "\n1.Enqueue 2.Dequeue 3.Display 4.Exit\n";
        cout << "Enter choice: ";
        cin >> choice;

        switch (choice) {
            case 1:
                cout << "Enter value: ";
                cin >> value;
                enqueue(value);
                break;
            case 2:
                dequeue();
                break;
            case 3:
                display();
                break;
            case 4:
                cout << "Exiting...";
                break;
            default:
                cout << "Invalid choice\n";
        }
    } while (choice != 4);

    return 0;
}

Sample Output

1.Enqueue 2.Dequeue 3.Display 4.Exit Enter choice: 1 Enter value: 10 Inserted: 10 Enter choice: 1 Enter value: 20 Inserted: 20 Enter choice: 3 Queue elements: 10 20 Enter choice: 2 Deleted: 10 Enter choice: 4 Exiting...

8. Program to Delete a Node at the End of a Singly Linked List

Aim: To delete the last node from a singly linked list.

Source Code

#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* next;
};

Node* head = NULL;

void insert(int value) {
    Node* newNode = new Node;
    newNode->data = value;
    newNode->next = NULL;

    if (head == NULL) {
        head = newNode;
        return;
    }

    Node* temp = head;
    while (temp->next != NULL)
        temp = temp->next;

    temp->next = newNode;
}

void deleteEnd() {
    if (head == NULL) {
        cout << "List is empty\n";
        return;
    }

    if (head->next == NULL) {
        delete head;
        head = NULL;
        return;
    }

    Node* temp = head;

    while (temp->next->next != NULL)
        temp = temp->next;

    Node* last = temp->next;
    cout << "Deleted: " << last->data << endl;

    delete last;
    temp->next = NULL;
}

void display() {
    Node* temp = head;
    cout << "Linked List: ";

    while (temp != NULL) {
        cout << temp->data << " -> ";
        temp = temp->next;
    }

    cout << "NULL\n";
}

int main() {
    int n, value;

    cout << "Enter number of nodes: ";
    cin >> n;

    for (int i = 0; i < n; i++) {
        cout << "Enter value: ";
        cin >> value;
        insert(value);
    }

    display();
    deleteEnd();
    display();

    return 0;
}

1. Program to Construct a Binary Search Tree

Aim: To construct a Binary Search Tree (BST) by inserting elements.

Source Code

#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* left;
    Node* right;
};

Node* createNode(int value) {
    Node* newNode = new Node;
    newNode->data = value;
    newNode->left = newNode->right = NULL;
    return newNode;
}

Node* insert(Node* root, int value) {
    if (root == NULL)
        return createNode(value);

    if (value < root->data)
        root->left = insert(root->left, value);
    else if (value > root->data)
        root->right = insert(root->right, value);

    return root;
}

void inorder(Node* root) {
    if (root != NULL) {
        inorder(root->left);
        cout << root->data << " ";
        inorder(root->right);
    }
}

int main() {
    Node* root = NULL;
    int n, value;

    cout << "Enter number of nodes: ";
    cin >> n;

    for (int i = 0; i < n; i++) {
        cout << "Enter value: ";
        cin >> value;
        root = insert(root, value);
    }

    cout << "BST Inorder: ";
    inorder(root);

    return 0;
}

Sample Output

Enter number of nodes: 5 Enter value: 40 Enter value: 20 Enter value: 60 Enter value: 10 Enter value: 30 BST Inorder: 10 20 30 40 60

Result: A Binary Search Tree is constructed successfully.

2. Program for Binary Tree Traversal

Aim: To perform inorder, preorder and postorder traversal of a binary tree.

Source Code

#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* left;
    Node* right;
};

Node* createNode(int value) {
    Node* node = new Node;
    node->data = value;
    node->left = node->right = NULL;
    return node;
}

void inorder(Node* root) {
    if (root != NULL) {
        inorder(root->left);
        cout << root->data << " ";
        inorder(root->right);
    }
}

void preorder(Node* root) {
    if (root != NULL) {
        cout << root->data << " ";
        preorder(root->left);
        preorder(root->right);
    }
}

void postorder(Node* root) {
    if (root != NULL) {
        postorder(root->left);
        postorder(root->right);
        cout << root->data << " ";
    }
}

int main() {
    Node* root = createNode(1);
    root->left = createNode(2);
    root->right = createNode(3);
    root->left->left = createNode(4);
    root->left->right = createNode(5);

    cout << "Inorder: ";
    inorder(root);

    cout << "\nPreorder: ";
    preorder(root);

    cout << "\nPostorder: ";
    postorder(root);

    return 0;
}

Sample Output

Inorder: 4 2 5 1 3 Preorder: 1 2 4 5 3 Postorder: 4 5 2 3 1

Result: All three binary tree traversal methods are performed successfully.

3. Program to Implement Depth First Search (DFS)

Aim: To traverse a graph using Depth First Search.

Source Code

#include <iostream>
using namespace std;

int graph[10][10], visited[10], n;

void DFS(int vertex) {
    cout << vertex << " ";
    visited[vertex] = 1;

    for (int i = 0; i < n; i++) {
        if (graph[vertex][i] == 1 && !visited[i]) {
            DFS(i);
        }
    }
}

int main() {
    int edges, u, v, start;

    cout << "Enter number of vertices: ";
    cin >> n;

    cout << "Enter number of edges: ";
    cin >> edges;

    for (int i = 0; i < edges; i++) {
        cout << "Enter edge (u v): ";
        cin >> u >> v;
        graph[u][v] = graph[v][u] = 1;
    }

    cout << "Enter starting vertex: ";
    cin >> start;

    cout << "DFS Traversal: ";
    DFS(start);

    return 0;
}

Sample Output

4. Program to Implement Breadth First Search (BFS)

Aim: To traverse a graph using Breadth First Search.

Source Code

#include <iostream>
#include <queue>
using namespace std;

int graph[10][10], visited[10], n;

void BFS(int start) {
    queue<int> q;

    visited[start] = 1;
    q.push(start);

    while (!q.empty()) {
        int vertex = q.front();
        q.pop();

        cout << vertex << " ";

        for (int i = 0; i < n; i++) {
            if (graph[vertex][i] == 1 && !visited[i]) {
                visited[i] = 1;
                q.push(i);
            }
        }
    }
}

int main() {
    int edges, u, v, start;

    cout << "Enter number of vertices: ";
    cin >> n;

    cout << "Enter number of edges: ";
    cin >> edges;

    for (int i = 0; i < edges; i++) {
        cout << "Enter edge (u v): ";
        cin >> u >> v;
        graph[u][v] = graph[v][u] = 1;
    }

    cout << "Enter starting vertex: ";
    cin >> start;

    cout << "BFS Traversal: ";
    BFS(start);

    return 0;
}

#include <iostream>
using namespace std;

int main() {
    int a[100], n, temp, minIndex;

    cout << "Enter number of elements: ";
    cin >> n;

    cout << "Enter elements: ";
    for (int i = 0; i < n; i++)
        cin >> a[i];

    for (int i = 0; i < n - 1; i++) {
        minIndex = i;

        for (int j = i + 1; j < n; j++) {
            if (a[j] < a[minIndex])
                minIndex = j;
        }

        temp = a[i];
        a[i] = a[minIndex];
        a[minIndex] = temp;
    }

    cout << "Sorted array: ";
    for (int i = 0; i < n; i++)
        cout << a[i] << " ";

    return 0;
}

8. Program to Perform Binary Search

Aim: To search for an element in a sorted array using Binary Search.

Source Code

#include <iostream>
using namespace std;

int main() {
    int a[100], n, key;
    int low, high, mid, found = 0;

    cout << "Enter number of elements: ";
    cin >> n;

    cout << "Enter sorted array elements: ";
    for (int i = 0; i < n; i++)
        cin >> a[i];

    cout << "Enter element to search: ";
    cin >> key;

    low = 0;
    high = n - 1;

    while (low <= high) {
        mid = low + (high - low) / 2;

        if (a[mid] == key) {
            cout << "Element found at position " << mid + 1;
            found = 1;
            break;
        }
        else if (a[mid] < key)
            low = mid + 1;
        else
            high = mid - 1;
    }

    if (!found)
        cout << "Element not found";

    return 0;
}

Sample Output

Enter number of elements: 5 Enter sorted array elements: 10 20 30 40 50 Enter element to search: 40 Element found at position 4

Result: An element is searched successfully using Binary Search.

Important Viva Questions

1. What is a data structure?

A data structure is a method of organizing and storing data so that it can be accessed and modified efficiently.

2. What is a stack?

A stack is a linear data structure that follows LIFO (Last In, First Out).

3. What is a queue?

A queue is a linear data structure that follows FIFO (First In, First Out).

4. What is a linked list?

A linked list is a collection of nodes where each node contains data and a link to the next node.

5. What is recursion?

Recursion is a technique in which a function calls itself.

6. What is a binary tree?

A binary tree is a tree in which each node can have at most two children.

7. What is a Binary Search Tree?

A BST is a binary tree in which smaller values are stored on the left and larger values on the right.

8. What is DFS?

Depth First Search explores a graph deeply along a path before backtracking.

9. What is BFS?

Breadth First Search explores graph vertices level by level using a queue.

10. What is sorting?

Sorting is the process of arranging data in ascending or descending order.

11. What is searching?

Searching is the process of finding a particular element in a collection.

12. What is the difference between Linear Search and Binary Search?

Linear Search checks elements one by one. Binary Search repeatedly divides a sorted search range into two halves.
